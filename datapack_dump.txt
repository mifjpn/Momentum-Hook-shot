   3: # 回転行列で実行方向へ座標を補正します

  10: 
  11: # Y` = Y * cos(pitch) - Z * sin(pitch)
  12:     scoreboard players operation $Y ForwardSpreader = $VecY ForwardSpreader
  13:     scoreboard players operation $Y ForwardSpreader *= $PitchCos ForwardSpreader
  14:     scoreboard players operation $Temp ForwardSpreader = $VecZ ForwardSpreader
  15:     scoreboard players operation $Temp ForwardSpreader *= $PitchSin ForwardSpreader
  16:     scoreboard players operation $Y ForwardSpreader -= $Temp ForwardSpreader
  17: # Z` = Y * sin(pitch) + Z * cos(pitch)
  18:     scoreboard players operation $Z ForwardSpreader = $VecY ForwardSpreader
  19:     scoreboard players operation $Z ForwardSpreader *= $PitchSin ForwardSpreader
  20:     scoreboard players operation $Temp ForwardSpreader = $VecZ ForwardSpreader

  22:     scoreboard players operation $Z ForwardSpreader += $Temp ForwardSpreader

  24: # X` = X * cos(yaw) + Z` * sin(yaw)

  26:     scoreboard players operation $X ForwardSpreader *= $YawCos ForwardSpreader
  27:     scoreboard players operation $Temp ForwardSpreader = $Z ForwardSpreader
  28:     scoreboard players operation $Temp ForwardSpreader *= $YawSin ForwardSpreader
  29:     scoreboard players operation $X ForwardSpreader += $Temp ForwardSpreader
  30: # Z` = Z` * cos(yaw) - X * sin(yaw)
  31:     scoreboard players operation $Z ForwardSpreader *= $YawCos ForwardSpreader
  32:     scoreboard players operation $Temp ForwardSpreader = $VecX ForwardSpreader
  33:     scoreboard players operation $Temp ForwardSpreader *= $YawSin ForwardSpreader
  34:     scoreboard players operation $Z ForwardSpreader -= $Temp ForwardSpreader
  35: # 係数が乗算で膨れ上がってるので除算
  36:     scoreboard players operation $X ForwardSpreader /= $10000 ForwardSpreader
  37:     scoreboard players operation $Y ForwardSpreader /= $10000 ForwardSpreader
  38:     scoreboard players operation $Z ForwardSpreader /= $10000 ForwardSpreader
